107 }

117int main(int argc, char **argv)
118{
119 try
120 {
121 const auto options = parseArgs(argc, argv);
122 stgs::TerminalUi ui(options.color);
123 ui.section("STGS PORT CHECK");
124 ui.keyValue("target", options.host);
125 ui.keyValue("range", std::to_string(options.ports.first) + ":" + std::to_string(options.ports.last));
126 ui.keyValue("timeout", std::to_string(options.timeout.count()) + " ms / port");
127
128 const auto results = stgs::scanTcpPorts(options.host, options.ports, options.timeout);
129 std::size_t openCount = 0U;
130 ui.section("REPORT");
131 for (const auto &result : results)
132 {
133 if (result.state == stgs::TcpPortState::Open)
134 {
135 ++openCount;
136 }
137 std::ostringstream latency;
138 latency << std::fixed << std::setprecision(2) << result.latencyMs << " ms";
139 stgs::TerminalStatus visualStatus = stgs::TerminalStatus::Info;
140 switch (result.state)
141 {
142 case stgs::TcpPortState::Open:
143 visualStatus = stgs::TerminalStatus::Success;
144 break;
145 case stgs::TcpPortState::Closed:
146 visualStatus = stgs::TerminalStatus::Info;
147 break;
148 case stgs::TcpPortState::Timeout:
149 visualStatus = stgs::TerminalStatus::Warning;
150 break;
151 case stgs::TcpPortState::Error:
152 visualStatus = stgs::TerminalStatus::Error;
153 break;
154 }
155 ui.statusLine(visualStatus,
156 "port " + std::to_string(result.port) + " / " +
157 stgs::tcpPortStateToString(result.state),
158 latency.str() + " / " + result.detail);
159 }
160
161 ui.section("SUMMARY");
162 ui.keyValue("ports tested", std::to_string(results.size()));
163 ui.keyValue("TCP listeners", std::to_string(openCount));
164 ui.keyValue("interpretation", "OPEN = TCP listener reachable; service identity is not inferred");
165 return 0;
166 }
167 catch (const std::exception &ex)
168 {
169 std::cerr << "error: " << ex.what() << '\n';
170 return 1;
171 }
