132 ApplicationPayloadParseResult decodeApplicationPayload(std::span<const std::uint8_t> payload)
133 {
134 if (!hasApplicationMagic(payload))
135 {
136 return std::monostate{};
137 }
138 if (payload.size() < ApplicationCommonHeaderSize)
139 {
140 return makeError(ApplicationPayloadErrorCode::TruncatedHeader,
141 "STGA signature is present but the common header is truncated");
142 }
143 if (payload[AppVersionOffset] != ApplicationPayloadVersion)
144 {
145 return makeError(ApplicationPayloadErrorCode::UnsupportedVersion,
146 "unsupported STGA application payload version");
147 }
148
149 const auto rawKind = payload[AppKindOffset];
150 if (rawKind == static_cast<std::uint8_t>(ApplicationPayloadKind::TextMessage))
151 {
152 if (payload.size() < TextBodyOffset)
153 {
154 return makeError(ApplicationPayloadErrorCode::InvalidLength,
155 "text message payload is shorter than its fixed header");
156 }
157 TextMessagePayload message;
158 message.sequence = detail::readU32BE(payload, TextSequenceOffset);
159 message.text.assign(reinterpret_cast<const char *>(payload.data() + TextBodyOffset),
160 payload.size() - TextBodyOffset);
161 return message;
162 }
163
164 if (rawKind == static_cast<std::uint8_t>(ApplicationPayloadKind::SignalBlock))
165 {
166 if (payload.size() < SignalSamplesOffset)
167 {
168 return makeError(ApplicationPayloadErrorCode::InvalidLength,
169 "signal payload is shorter than its fixed header");
170 }
171
172 SignalBlockPayload signal;
173 signal.sampleRateHz = detail::readU16BE(payload, SignalSampleRateOffset);
174 signal.frequencyHz = detail::readFloatBE(payload, SignalFrequencyOffset);
175 signal.nominalAmplitude = detail::readFloatBE(payload, SignalAmplitudeOffset);
176 signal.startSampleIndex = detail::readU32BE(payload, SignalStartIndexOffset);
177 const auto sampleCount = detail::readU16BE(payload, SignalSampleCountOffset);
178 const auto expectedSize = SignalSamplesOffset + static_cast<std::size_t>(sampleCount) * SignalSampleWireSize;
179 if (payload.size() != expectedSize)
180 {
181 return makeError(ApplicationPayloadErrorCode::InvalidLength,
182 "signal sample count does not match the payload length");
183 }
184 if (!validSignalMetadata(signal.sampleRateHz, signal.frequencyHz, signal.nominalAmplitude))
185 {
186 return makeError(ApplicationPayloadErrorCode::InvalidSignalMetadata,
187 "signal metadata requires finite amplitude, sampleRate > 0 and 0 < frequency < Nyquist");
188 }
189
190 signal.samples.reserve(sampleCount);
191 for (std::size_t i = 0; i < sampleCount; ++i)
192 {
193 const auto offset = SignalSamplesOffset + i * SignalSampleWireSize;
194 const float sample = detail::readFloatBE(payload, offset);
195 if (!std::isfinite(sample))
196 {
197 return makeError(ApplicationPayloadErrorCode::InvalidSignalMetadata,
198 "signal payload contains a non-finite sample");
199 }
200 signal.samples.push_back(sample);
201 }
202 return signal;
203 }
204
205 return makeError(ApplicationPayloadErrorCode::UnknownKind, "unknown STGA application payload kind");
206 }
