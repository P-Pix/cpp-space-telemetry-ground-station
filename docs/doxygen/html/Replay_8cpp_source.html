124 std::optional<ByteVector> FrameFileReader::readNext()
125 {
126 std::array<std::uint8_t, ReplayLengthFieldSize> lengthBytes{};
127 in_.read(reinterpret_cast<char *>(lengthBytes.data()),
128 static_cast<std::streamsize>(lengthBytes.size()));
129 const auto bytesRead = in_.gcount();
130 if (bytesRead == 0 && in_.eof())
131 {
132 return std::nullopt;
133 }
134 if (bytesRead != static_cast<std::streamsize>(lengthBytes.size()))
135 {
136 throw std::runtime_error("truncated frame length in replay file");
137 }
138
139 const auto length = detail::readU32BE(lengthBytes, 0U);
140 if (length < MinFrameSize || length > MaxFrameSize)
141 {
142 throw std::runtime_error("invalid replay frame length");
143 }
144
145 ByteVector frame(length);
146 readExact(in_, frame.data(), frame.size());
147 return frame;
