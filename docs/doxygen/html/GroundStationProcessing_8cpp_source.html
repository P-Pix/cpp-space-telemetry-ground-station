49 {
50 ui.receivedMessage(frame.satelliteId, message->sequence, message->text);
51 return;
52 }
53 if (const auto *signal = std::get_if<SignalBlockPayload>(&parsed))
54 {
55 std::vector<float> filtered;
56 switch (options.signalFilter)

58 case SignalFilterMode::None:
59 filtered = signal->samples;
60 break;
61 case SignalFilterMode::MovingAverage:
62 filtered = movingAverageFilter(signal->samples, options.movingAverageWindow);
63 break;
64 case SignalFilterMode::SineProjection:
65 filtered = sineProjectionFilter(signal->samples,
66 signal->sampleRateHz,
67 signal->frequencyHz,
68 signal->startSampleIndex);
69 break;
70 }
71
72 const auto metrics = computeSignalMetrics(signal->samples,
73 filtered,
74 signal->sampleRateHz,
75 signal->frequencyHz,
76 signal->startSampleIndex);
77 ui.receivedSignal(frame.satelliteId,
78 signal->sampleRateHz,
79 signal->frequencyHz,
80 signal->nominalAmplitude,

82 filtered,
83 metrics,
