52 ProjectionCoefficients estimateProjection(std::span<const float> samples,
53 std::uint16_t sampleRateHz,
54 float frequencyHz,
55 std::uint32_t startSampleIndex)
56 {
57 ProjectionCoefficients result;
58 if (samples.empty())
59 {
60 return result;
61 }
62 validateProjectionMetadata(sampleRateHz, frequencyHz);
63 validateFiniteSamples(samples);
64
65 double sum = 0.0;
66 double sinProjection = 0.0;
67 double cosProjection = 0.0;
68 const double omega = TwoPi * static_cast<double>(frequencyHz) / static_cast<double>(sampleRateHz);
69 for (std::size_t i = 0; i < samples.size(); ++i)
70 {
71 const double x = static_cast<double>(samples[i]);
72 const double phase = omega * static_cast<double>(static_cast<std::uint64_t>(startSampleIndex) + i);
73 sum += x;
74 sinProjection += x * std::sin(phase);
75 cosProjection += x * std::cos(phase);
76 }
77
78 const double count = static_cast<double>(samples.size());
79 result.dc = sum / count;
80 // Pour un bloc couvrant plusieurs périodes, sin et cos ont une énergie proche de N/2.
81 // Le facteur 2/N transforme donc directement les projections en amplitudes estimées.
82 result.sinCoefficient = 2.0 * sinProjection / count;
83 result.cosCoefficient = 2.0 * cosProjection / count;
84 return result;
85 }

87 double rootMeanSquare(std::span<const float> samples) noexcept
88 {
89 if (samples.empty())
90 {
91 return 0.0;
92 }
93 double sumSquares = 0.0;
94 for (const float sample : samples)
95 {
96 const double x = static_cast<double>(sample);
97 sumSquares += x * x;
98 }
99 return std::sqrt(sumSquares / static_cast<double>(samples.size()));

104 std::vector<float> movingAverageFilter(std::span<const float> samples, std::size_t windowSize)
105 {
106 if (windowSize == 0U || (windowSize % 2U) == 0U)
107 {
108 throw std::invalid_argument("moving-average window must be a positive odd number");
109 }
110 validateFiniteSamples(samples);
111 if (samples.empty() || windowSize == 1U)
112 {
113 return std::vector<float>(samples.begin(), samples.end());
114 }
115
116 const std::size_t radius = windowSize / 2U;
117 std::vector<float> filtered(samples.size(), 0.0F);
118 for (std::size_t i = 0; i < samples.size(); ++i)
119 {
120 const std::size_t begin = i > radius ? i - radius : 0U;
121 const std::size_t end = std::min(samples.size(), i + radius + 1U);
122 double sum = 0.0;
123 for (std::size_t j = begin; j < end; ++j)
124 {
125 sum += static_cast<double>(samples[j]);
126 }
127 filtered[i] = static_cast<float>(sum / static_cast<double>(end - begin));
128 }
129 return filtered;
130 }

132 std::vector<float> sineProjectionFilter(std::span<const float> samples,
133 std::uint16_t sampleRateHz,
134 float frequencyHz,
135 std::uint32_t startSampleIndex)
136 {
137 const auto coefficients = estimateProjection(samples, sampleRateHz, frequencyHz, startSampleIndex);
138 std::vector<float> filtered(samples.size(), 0.0F);
139 if (samples.empty())
140 {
141 return filtered;
142 }
143
144 const double omega = TwoPi * static_cast<double>(frequencyHz) / static_cast<double>(sampleRateHz);
145 for (std::size_t i = 0; i < samples.size(); ++i)
146 {
147 const double phase = omega * static_cast<double>(static_cast<std::uint64_t>(startSampleIndex) + i);
148 const double value = coefficients.dc +
149 coefficients.sinCoefficient * std::sin(phase) +
150 coefficients.cosCoefficient * std::cos(phase);
151 filtered[i] = static_cast<float>(value);
152 }
153 return filtered;
154 }

156 SignalMetrics computeSignalMetrics(std::span<const float> raw,
157 std::span<const float> filtered,
158 std::uint16_t sampleRateHz,
159 float frequencyHz,
160 std::uint32_t startSampleIndex)
161 {
162 if (raw.size() != filtered.size())
163 {
164 throw std::invalid_argument("raw and filtered signal sizes must match");
165 }
166 validateProjectionMetadata(sampleRateHz, frequencyHz);
167 validateFiniteSamples(raw);
168 validateFiniteSamples(filtered);
169
170 SignalMetrics metrics;
171 metrics.rawRms = rootMeanSquare(raw);
172 metrics.filteredRms = rootMeanSquare(filtered);
173
174 if (!raw.empty())
175 {
176 double residualSquares = 0.0;
177 for (std::size_t i = 0; i < raw.size(); ++i)
178 {
179 const double residual = static_cast<double>(raw[i]) - static_cast<double>(filtered[i]);
180 residualSquares += residual * residual;
181 }
182 metrics.residualNoiseRms = std::sqrt(residualSquares / static_cast<double>(raw.size()));
183 }
184
185 const auto coefficients = estimateProjection(raw, sampleRateHz, frequencyHz, startSampleIndex);
186 metrics.estimatedAmplitude = std::hypot(coefficients.sinCoefficient, coefficients.cosCoefficient);
187
188 // Seuil purement numérique : évite log10(0) et les divisions instables quand un résidu est
189 // nul à la précision double. Il ne représente pas un seuil physique de bruit ou de sensibilité.
190 constexpr double NumericalZeroEpsilon = 1.0e-12;
191 if (metrics.residualNoiseRms <= NumericalZeroEpsilon)
192 {
193 metrics.estimatedSnrDb = std::numeric_limits<double>::infinity();
194 }
195 else if (metrics.filteredRms <= NumericalZeroEpsilon)
196 {
197 metrics.estimatedSnrDb = -std::numeric_limits<double>::infinity();
198 }
199 else
200 {
201 metrics.estimatedSnrDb = DecibelsPerAmplitudeDecade *
202 std::log10(metrics.filteredRms / metrics.residualNoiseRms);
203 }
204 return metrics;
205 }
