103 int run(Options options)
104 {
105 TerminalUi ui(options.color);
106 discoverPort(options, ui);
107 renderConfiguration(options, ui);
108
109 std::mt19937 rng(options.seed);
110 std::uniform_real_distribution<double> probability(0.0, 1.0);
111 std::unique_ptr<FrameFileWriter> capture;
112 if (options.outputFile.has_value())
113 {
114 capture = std::make_unique<FrameFileWriter>(*options.outputFile);
115 }
116
117 std::optional<SocketFd> socket;
118 sockaddr_in address{};
119 if (options.transport.has_value())
120 {
121 socket.emplace(createSocket(*options.transport));
122 address = destination(options);
123 if (*options.transport == Transport::Tcp)
124 {
125 ui.section("TCP CONNECTION");
126 ui.keyValue("status", "connecting...");
127 if (::connect(socket->get(), reinterpret_cast<const sockaddr *>(&address), sizeof(address)) < 0)
128 {
129 throw std::runtime_error("connect() failed: " + std::string(std::strerror(errno)));
130 }
131 ui.keyValue("status", "connected");
132 }
133 }
134
135 const auto delay = options.rate > 0.0
136 ? std::chrono::duration<double>(1.0 / options.rate)
137 : std::chrono::duration<double>(0.0);
138 auto nextTick = std::chrono::steady_clock::now();
139
140 std::size_t produced = 0U;
141 std::size_t dropped = 0U;
142 std::size_t corrupted = 0U;
143 ui.section("TRANSMISSION");
144 for (std::size_t i = 0U; i < options.count; ++i)
145 {
146 auto frame = makeFrame(options, rng, i);
147 auto bytes = encodeFrame(frame);
148
149 if (probability(rng) < options.lossProbability)
150 {
151 ++dropped;
152 }
153 else
154 {
155 if (probability(rng) < options.corruptProbability)
156 {
157 maybeCorrupt(bytes, rng);
158 ++corrupted;
159 }
160
161 if (socket.has_value())
162 {
163 if (*options.transport == Transport::Udp)
164 {
165 sendUdpFrame(socket->get(), address, bytes);
166 }
167 else
168 {
169 sendAll(socket->get(), bytes);
170 }
171 }
172 if (capture)
173 {
174 capture->writeFrame(bytes);
175 }
176 ++produced;
177 }
178
179 if (delay.count() > 0.0)
180 {
181 nextTick += std::chrono::duration_cast<std::chrono::steady_clock::duration>(delay);
182 std::this_thread::sleep_until(nextTick);
183 }
184 }
185
186 if (capture)
187 {
188 capture->flush();
189 }
190
191 ui.section("SIMULATOR SUMMARY");
192 ui.keyValue("produced", std::to_string(produced));
193 ui.keyValue("dropped", std::to_string(dropped));
194 ui.keyValue("corrupted", std::to_string(corrupted));
195 if (options.outputFile.has_value())
196 {
197 ui.keyValue("replay", options.outputFile->string());
198 }
199 return 0;
200 }
