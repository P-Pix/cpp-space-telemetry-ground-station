162 inline PortRange parsePortRange(std::string_view value, std::string_view name)
163 {
164 const auto separator = value.find(':');
165 if (separator == std::string_view::npos)
166 {
167 const auto port = parseUnsigned<std::uint16_t>(value, name, MinimumNetworkPort, MaximumNetworkPort);
168 return PortRange{port, port};
169 }
170 if (value.find(':', separator + 1U) != std::string_view::npos)
171 {
172 throw std::runtime_error(std::string(name) + " must use start:end syntax");
173 }
174 const auto first = parseUnsigned<std::uint16_t>(value.substr(0U, separator), name, MinimumNetworkPort, MaximumNetworkPort);
175 const auto last = parseUnsigned<std::uint16_t>(value.substr(separator + 1U), name, MinimumNetworkPort, MaximumNetworkPort);
176 if (first > last)
177 {
178 throw std::runtime_error(std::string(name) + " start port cannot exceed end port");
179 }
180 return PortRange{first, last};
