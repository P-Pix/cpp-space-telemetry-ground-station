168 std::string TerminalUi::sanitizeRemoteText(std::string_view text)
169 {
170 std::ostringstream out;
171 for (const char raw : text)
172 {
173 const auto ch = static_cast<unsigned char>(raw);
174 switch (ch)
175 {
176 case '\n':
177 out << "\\n";
178 break;
179 case '\r':
180 out << "\\r";
181 break;
182 case '\t':
183 out << "\\t";
184 break;
185 case AsciiEscape: // ESC : neutralise explicitement les séquences de contrôle ANSI distantes.
186 out << "\\x1B";
187 break;
188 default:
189 if (ch < AsciiControlUpperExclusive || ch == AsciiDelete)
190 {
191 out << "\\x" << std::uppercase << std::hex << std::setw(EscapedByteHexWidth) << std::setfill('0')
192 << static_cast<unsigned int>(ch) << std::dec << std::nouppercase;
193 }
194 else
195 {
196 out << static_cast<char>(ch);
197 }
198 break;
199 }
200 }
201 return out.str();
202 }

215 std::string TerminalUi::sparkline(std::span<const float> samples, std::size_t columns, bool unicode)
216 {
217 if (samples.empty() || columns == 0U)
218 {
219 return {};
220 }
221
222 const auto [minIt, maxIt] = std::minmax_element(samples.begin(), samples.end());
223 const double minimum = static_cast<double>(*minIt);
224 const double maximum = static_cast<double>(*maxIt);
225 const double range = maximum - minimum;
226
227 const std::array<std::string_view, SparklineLevelCount> unicodeLevels{"▁", "▂", "▃", "▄", "▅", "▆", "▇", "█"};
228 constexpr std::string_view asciiLevels = "._-~=+*#";
229
230 std::string out;
231 const std::size_t outputColumns = std::min(columns, samples.size());
232 for (std::size_t column = 0; column < outputColumns; ++column)
233 {
234 const std::size_t begin = column * samples.size() / outputColumns;
235 const std::size_t end = std::max(begin + 1U, (column + 1U) * samples.size() / outputColumns);
236 double average = 0.0;
237 for (std::size_t i = begin; i < end; ++i)
238 {
239 average += static_cast<double>(samples[i]);
240 }
241 average /= static_cast<double>(end - begin);
242 const double normalized = range <= std::numeric_limits<double>::epsilon()
243 ? FlatSignalNormalizedLevel
244 : std::clamp((average - minimum) / range, 0.0, 1.0);
245 const auto level = static_cast<std::size_t>(
246 std::lround(normalized * static_cast<double>(SparklineLevelCount - 1U)));
247 if (unicode)
248 {
249 out += unicodeLevels[level];
250 }
251 else
252 {
253 out.push_back(asciiLevels[level]);
254 }
255 }
256 return out;
257 }
