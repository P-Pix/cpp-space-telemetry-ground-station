85 std::optional<StationStateTransition> StationHealthMonitor::recordSample(Sample sample)
86 {
87 std::lock_guard<std::mutex> lock(mutex_);
88 if (!config_.enabled)
89 {
90 return std::nullopt;
91 }
92
93 samples_.push_back(sample);
94 while (samples_.size() > config_.windowSize)
95 {
96 samples_.pop_front();
97 }
98
99 const auto snapshot = makeSnapshotLocked();
100 if (snapshot.samples < config_.minSamples)
101 {
102 return std::nullopt;
103 }
104
105 StationState target = state_;
106 if (state_ == StationState::Nominal)
107 {
108 if (snapshot.rejectionRate >= config_.degradedRejectionRate ||
109 snapshot.critical >= config_.criticalFramesForDegraded)
110 {
111 target = StationState::Degraded;
112 }
113 }
114 else
115 {
116 if (snapshot.rejectionRate <= config_.recoveryRejectionRate &&
117 snapshot.critical <= config_.criticalFramesForRecovery)
118 {
119 target = StationState::Nominal;
120 }
121 }
122
123 if (target == state_)
124 {
125 return std::nullopt;
126 }
127
128 StationStateTransition transition;
129 transition.from = state_;
130 transition.to = target;
131 transition.snapshot = snapshot;
132 transition.reason = transitionReasonLocked(snapshot, target);
133 state_ = target;
134 transition.snapshot.state = state_;
135 return transition;
136 }

150 StationHealthSnapshot StationHealthMonitor::makeSnapshotLocked() const
151 {
152 StationHealthSnapshot snapshot;
153 snapshot.state = state_;
154 snapshot.samples = samples_.size();
155 for (const auto &sample : samples_)
156 {
157 if (sample.rejected)
158 {
159 ++snapshot.rejected;
160 }
161 if (sample.critical)
162 {
163 ++snapshot.critical;
164 }
165 }
166 if (snapshot.samples > 0U)

168 snapshot.rejectionRate = static_cast<double>(snapshot.rejected) / static_cast<double>(snapshot.samples);
