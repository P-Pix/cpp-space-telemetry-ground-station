121 serverFd_ = createBoundSocket(SOCK_STREAM);
122 setNonBlocking(serverFd_);
123 if (::listen(serverFd_, SOMAXCONN) < 0)

127 }
128
129 logger_.info("TCP receiver listening on " + config_.bindAddress + ":" + std::to_string(config_.port));
130 std::vector<TcpClient> clients;
131 ByteVector readBuffer(TcpReadBufferSize);
132
133 while (running.load() && !stopRequested_.load())
134 {
135 const std::size_t polledClientCount = clients.size();
136 std::vector<pollfd> pfds;
137 pfds.reserve(polledClientCount + 1U);
138 pfds.push_back(pollfd{serverFd_, POLLIN, 0});
139 for (std::size_t i = 0U; i < polledClientCount; ++i)
140 {
141 pfds.push_back(pollfd{clients[i].fd, POLLIN | POLLHUP | POLLERR, 0});
142 }
143
144 const int rc = ::poll(pfds.data(), static_cast<nfds_t>(pfds.size()), config_.pollTimeoutMs);
145 if (rc < 0)
146 {
147 if (errno == EINTR)
148 {
149 continue;
150 }
151 throw std::runtime_error("poll() failed: " + std::string(std::strerror(errno)));
152 }
153 if (rc == 0)
154 {
155 continue;
156 }
157
158 for (std::size_t snapshotIndex = polledClientCount; snapshotIndex > 0U; --snapshotIndex)
159 {
160 const std::size_t clientIndex = snapshotIndex - 1U;
161 const short revents = pfds[snapshotIndex].revents;
162 if (revents == 0)
163 {
164 continue;
165 }
166
167 TcpClient &client = clients[clientIndex];
168 bool removeClient = (revents & (POLLERR | POLLNVAL)) != 0;
169 if ((revents & (POLLIN | POLLHUP)) != 0)
170 {
171 while (true)
172 {
173 const ssize_t count = ::recv(client.fd, readBuffer.data(), readBuffer.size(), 0);
174 if (count > 0)
175 {
176 const auto bytes = std::span<const std::uint8_t>(
177 readBuffer.data(), static_cast<std::size_t>(count));
178 auto frames = client.extractor.feed(bytes);
179 for (auto &frame : frames)
180 {
181 callback(std::move(frame));
182 }
183 continue;
184 }
185 if (count == 0)
186 {
187 removeClient = true;
188 break;
189 }
190 if (errno == EINTR)
191 {
192 continue;
193 }
194 if (errno == EAGAIN || errno == EWOULDBLOCK)
195 {
196 break;
197 }
198 logger_.warning("recv() failed: " + std::string(std::strerror(errno)));
199 removeClient = true;
200 break;
201 }
202 }
203
204 if ((revents & POLLHUP) != 0)
205 {
206 removeClient = true;
207 }
208 if (removeClient)
209 {
210 logger_.info("TCP client disconnected");
211 clients.erase(clients.begin() + static_cast<std::ptrdiff_t>(clientIndex));
212 }
213 }
214
215 if ((pfds.front().revents & POLLIN) != 0)
216 {
217 while (true)
218 {
219 sockaddr_in peer{};
220 socklen_t peerLength = sizeof(peer);
221 ScopedAcceptedFd accepted(::accept(serverFd_, reinterpret_cast<sockaddr *>(&peer), &peerLength));
222 if (accepted.get() < 0)
223 {
224 if (errno == EAGAIN || errno == EWOULDBLOCK)
225 {
226 break;
227 }
228 if (errno == EINTR)
229 {
230 continue;
231 }
232 logger_.warning("accept() failed: " + std::string(std::strerror(errno)));
233 break;
234 }
235
236 setNonBlocking(accepted.get());
237 clients.emplace_back(accepted.release());
238 logger_.info("TCP client connected");
239 }
240 }
241 }
