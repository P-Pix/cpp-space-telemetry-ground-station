27 void selectAutomaticPort(Options &options, TerminalUi &ui)
28 {
29 if (!options.autoPortRange.has_value())
30 {
31 return;
32 }
33
34 ui.section("LOCAL PORT SELECTION");
35 std::vector<LocalPortAvailability> report;
36 const auto selected = findFirstAvailableLocalPort(
37 options.bindAddress, *options.autoPortRange, PortTransport::Tcp, &report);
38 for (const auto &entry : report)
39 {
40 ui.keyValue("port " + std::to_string(entry.port),
41 std::string(entry.available ? "AVAILABLE" : "BUSY/ERROR") + " / " + entry.detail);
42 }
43 if (!selected.has_value())
44 {
45 throw std::runtime_error("no bindable port found in --auto-port range");
46 }
47 options.port = *selected;
48 ui.keyValue("selected", std::to_string(options.port));
49 }

108 int run(Options options)
109 {
110 TerminalUi ui(options.color);
111 selectAutomaticPort(options, ui);
112
113 Logger logger(options.logFile, options.logLevel);
114 StationHealthMonitor health(options.healthConfig);
115 std::atomic_bool running{true};
116 PosixSignalStopController signalStopController(running);
117
118 renderStartup(options, ui);
119 const auto outputFormat = resolveOutputFormat(options.outputFormat, options.outputFile);
120 GroundStationPipeline pipeline(options, outputFormat, ui, logger, health, running);
121 pipeline.run();
122
123 renderSummary(options, pipeline.stats(), health.snapshot(), ui, logger);
124 signalStopController.finish();
125 return 0;
126 }
