88 void GroundStationPipeline::writerLoop()
89 {

92 std::uint64_t expectedSequence = 0U;
93 std::map<std::uint64_t, DecodedWorkItem> pending;
94 std::uint64_t localApplicationErrors = 0U;

96 while (auto incoming = decodedQueue_.pop())
97 {
98 const auto [iterator, inserted] = pending.emplace(incoming->sequence, std::move(*incoming));
99 if (!inserted)
100 {
101 throw std::runtime_error("duplicate pipeline sequence detected");
102 }
103 (void)iterator;
104
105 while (true)
106 {
107 auto current = pending.find(expectedSequence);
108 if (current == pending.end())
109 {
110 break;
111 }
112
113 if (auto *frame = std::get_if<TelemetryFrame>(&current->second.result))
114 {
115 ++decoded_;
116 if (auto transition = health_.recordDecoded(*frame); transition.has_value())
117 {
118 logTransition(logger_, *transition);
119 }
120 processApplicationPayload(options_, *frame, ui_, logger_, localApplicationErrors);
121 output_.write(*frame);
122 ++written_;
123 if (options_.verbose)

126 }
127 }
128 else
129 {
130 ++rejected_;
131 const auto &error = std::get<FrameError>(current->second.result);
132 if (auto transition = health_.recordRejected(); transition.has_value())
133 {
134 logTransition(logger_, *transition);

136 logger_.warning("rejected seq=" + std::to_string(expectedSequence) + ": " +
137 errorCodeToString(error.code) + " - " + error.message);
