91 ByteVector encodeFrame(const TelemetryFrame &frame)
92 {
93 if (frame.payload.size() > MaxPayloadSize)
94 {
95 throw std::invalid_argument("payload exceeds MaxPayloadSize");
96 }
97 if (frame.version != ProtocolVersion)
98 {
99 throw std::invalid_argument("unsupported protocol version");
100 }
101 if (!std::isfinite(frame.temperatureC))
102 {
103 throw std::invalid_argument("temperature must be finite");
104 }
105 if (frame.batteryPercent > MaxBatteryPercent)
106 {
107 throw std::invalid_argument("battery percent must be between 0 and 100");
108 }
109
110 const auto statusByte = static_cast<std::uint8_t>(frame.status);
111 if (statusByte > static_cast<std::uint8_t>(Status::SafeMode))
112 {
113 throw std::invalid_argument("invalid status value");
114 }
115
116 ByteVector bytes;
117 bytes.reserve(HeaderSize + frame.payload.size() + CrcSize);
118 detail::appendU32BE(bytes, FrameMagic);
119 detail::appendU8(bytes, frame.version);
120 detail::appendU16BE(bytes, frame.satelliteId);
121 detail::appendU64BE(bytes, frame.timestampMs);
122 detail::appendFloatBE(bytes, frame.temperatureC);
123 detail::appendU8(bytes, frame.batteryPercent);
124 detail::appendU8(bytes, statusByte);
125 detail::appendU16BE(bytes, static_cast<std::uint16_t>(frame.payload.size()));
126 bytes.insert(bytes.end(), frame.payload.begin(), frame.payload.end());
127
128 const auto crc = crc32(bytes);
129 detail::appendU32BE(bytes, crc);
130 return bytes;
131 }

141 FrameParseResult decodeFrame(std::span<const std::uint8_t> bytes)
142 {
143 if (bytes.size() < MinFrameSize)
144 {
145 return makeError(FrameErrorCode::TooShort, "frame is shorter than the minimum header + CRC size");
146 }
147
148 const auto magic = detail::readU32BE(bytes, MagicOffset);
149 if (magic != FrameMagic)
150 {
151 return makeError(FrameErrorCode::BadMagic, "invalid frame magic; expected ASCII STGS");
152 }
153
154 const auto version = bytes[VersionOffset];
155 if (version != ProtocolVersion)
156 {
157 return makeError(FrameErrorCode::UnsupportedVersion, "unsupported protocol version");
158 }
159
160 const auto payloadLength = detail::readU16BE(bytes, PayloadLengthOffset);
161 if (payloadLength > MaxPayloadSize)
162 {
163 return makeError(FrameErrorCode::PayloadTooLarge, "payload length exceeds configured maximum");
164 }
165
166 const auto expectedSize = HeaderSize + static_cast<std::size_t>(payloadLength) + CrcSize;
167 if (bytes.size() != expectedSize)
168 {
169 std::ostringstream oss;
170 oss << "frame length mismatch: expected " << expectedSize << " bytes, got " << bytes.size();
171 return makeError(FrameErrorCode::LengthMismatch, oss.str());
172 }
173
174 const auto expectedCrc = detail::readU32BE(bytes, bytes.size() - CrcSize);
175 const auto calculatedCrc = crc32(bytes.subspan(0U, bytes.size() - CrcSize));
176 if (expectedCrc != calculatedCrc)
177 {
178 std::ostringstream oss;
179 oss << "CRC mismatch: expected 0x" << std::hex << std::setw(HexCrc32FieldWidth) << std::setfill('0') << expectedCrc
180 << ", calculated 0x" << std::setw(HexCrc32FieldWidth) << calculatedCrc;
181 return makeError(FrameErrorCode::BadCrc, oss.str());
182 }
183
184 const float temperature = detail::readFloatBE(bytes, TemperatureOffset);
185 if (!std::isfinite(temperature))
186 {
187 return makeError(FrameErrorCode::InvalidTemperature, "temperature field must be finite");
188 }
189
190 const auto battery = bytes[BatteryOffset];
191 if (battery > MaxBatteryPercent)
192 {
193 return makeError(FrameErrorCode::InvalidBattery, "battery percent is outside the 0..100 range");
194 }
195
196 const auto statusByte = bytes[StatusOffset];
197 if (statusByte > static_cast<std::uint8_t>(Status::SafeMode))
198 {
199 return makeError(FrameErrorCode::InvalidStatus, "status field contains an unknown value");
200 }
201
202 TelemetryFrame frame;
203 frame.version = version;
204 frame.satelliteId = detail::readU16BE(bytes, SatelliteIdOffset);
205 frame.timestampMs = detail::readU64BE(bytes, TimestampOffset);
206 frame.temperatureC = temperature;
207 frame.batteryPercent = battery;
208 frame.status = static_cast<Status>(statusByte);
209 frame.payload.assign(bytes.begin() + static_cast<std::ptrdiff_t>(HeaderSize),
210 bytes.begin() + static_cast<std::ptrdiff_t>(HeaderSize + payloadLength));
211
212 return frame;
213 }

250 std::vector<ByteVector> StreamFrameExtractor::feed(std::span<const std::uint8_t> bytes)
251 {
252 buffer_.insert(buffer_.end(), bytes.begin(), bytes.end());
253 std::vector<ByteVector> frames;
254 const auto magic = magicBytes();
255
256 while (true)
257 {
258 const auto it = std::search(buffer_.begin(), buffer_.end(), magic.begin(), magic.end());
259 if (it == buffer_.end())
260 {
261 if (buffer_.size() > MagicSize - 1U)
262 {
263 buffer_.erase(buffer_.begin(), buffer_.end() - static_cast<std::ptrdiff_t>(MagicSize - 1U));
264 }
265 return frames;
266 }
267
268 if (it != buffer_.begin())
269 {
270 buffer_.erase(buffer_.begin(), it);
271 }
272
273 if (buffer_.size() < HeaderSize)
274 {
275 return frames;
276 }
277
278 const auto payloadLength = detail::readU16BE(buffer_, PayloadLengthOffset);
279 if (payloadLength > MaxPayloadSize)
280 {
281 buffer_.erase(buffer_.begin());
282 continue;
283 }
284
285 const auto frameSize = HeaderSize + static_cast<std::size_t>(payloadLength) + CrcSize;
286 if (buffer_.size() < frameSize)
287 {
288 return frames;
289 }
290
291 frames.emplace_back(buffer_.begin(), buffer_.begin() + static_cast<std::ptrdiff_t>(frameSize));
292 buffer_.erase(buffer_.begin(), buffer_.begin() + static_cast<std::ptrdiff_t>(frameSize));
293 }
294 }
