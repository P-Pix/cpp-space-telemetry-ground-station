70 ByteVector makeSignalPayload(const Options &options, std::mt19937 &rng, std::size_t blockIndex)
71 {
72 const std::uint64_t startIndex64 = static_cast<std::uint64_t>(blockIndex) *
73 static_cast<std::uint64_t>(options.signalSamples);
74 if (startIndex64 > std::numeric_limits<std::uint32_t>::max())
75 {
76 throw std::runtime_error("signal sample index exceeds STGA v1 32-bit range");
77 }
78
79 std::normal_distribution<float> noise(0.0F, options.noiseStddev);
80 SignalBlockPayload signal;
81 signal.sampleRateHz = options.sampleRateHz;
82 signal.frequencyHz = options.signalFrequencyHz;
83 signal.nominalAmplitude = options.signalAmplitude;
84 signal.startSampleIndex = static_cast<std::uint32_t>(startIndex64);
85 signal.samples.reserve(options.signalSamples);
86
87 const double angularStep = TwoPi * static_cast<double>(options.signalFrequencyHz) /
88 static_cast<double>(options.sampleRateHz);
89 for (std::size_t i = 0U; i < options.signalSamples; ++i)
90 {
91 const std::uint64_t absoluteIndex = startIndex64 + static_cast<std::uint64_t>(i);
92 const double clean = static_cast<double>(options.signalAmplitude) *
93 std::sin(angularStep * static_cast<double>(absoluteIndex));
94 signal.samples.push_back(static_cast<float>(clean) + noise(rng));
95 }
96 return encodeSignalBlockPayload(signal);
97 }

101 TelemetryFrame makeFrame(const Options &options, std::mt19937 &rng, std::size_t sequence)
102 {
103 std::normal_distribution<float> temperature(
104 SimulationTemperatureMeanC, SimulationTemperatureStddevC);
105
106 TelemetryFrame frame;
107 frame.satelliteId = options.satelliteId;
108 frame.timestampMs = nowMs();
109 frame.temperatureC = temperature(rng);
110 frame.batteryPercent = static_cast<std::uint8_t>(
111 static_cast<std::size_t>(MaxBatteryPercent) - (sequence % SimulationBatteryCycleLength));
112 frame.status = randomStatus(rng);
113
114 switch (options.payloadMode)
115 {
116 case PayloadMode::RandomBytes:
117 frame.payload = makeRandomPayload(options, rng);
118 break;
119 case PayloadMode::TextMessage:
120 frame.payload = encodeTextMessagePayload(
121 TextMessagePayload{static_cast<std::uint32_t>(sequence), options.message});
122 break;
123 case PayloadMode::Signal:
124 frame.payload = makeSignalPayload(options, rng, sequence);
125 break;
126 }
127 return frame;
128 }
