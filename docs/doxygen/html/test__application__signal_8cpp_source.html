77 void testSignalProjectionRecoversKnownSine()
78 {
79 constexpr std::uint16_t SampleRate = 200U;
80 constexpr float Frequency = 5.0F;
81 constexpr std::size_t SampleCount = 400U;
82 constexpr double TwoPi = 2.0 * std::numbers::pi_v<double>;
83 std::vector<float> raw;
84 raw.reserve(SampleCount);
85 for (std::size_t i = 0U; i < SampleCount; ++i)
86 {
87 const double clean = std::sin(TwoPi * static_cast<double>(Frequency) *
88 static_cast<double>(i) / static_cast<double>(SampleRate));
89 const double deterministicNoise = (i % 2U == 0U) ? 0.35 : -0.35;
90 raw.push_back(static_cast<float>(clean + deterministicNoise));
91 }
92
93 const auto filtered = stgs::sineProjectionFilter(raw, SampleRate, Frequency, 0U);
94 const auto metrics = stgs::computeSignalMetrics(raw, filtered, SampleRate, Frequency, 0U);
95 assertNear(metrics.estimatedAmplitude, 1.0, 0.05, "estimated sine amplitude");
96 ASSERT_TRUE(metrics.residualNoiseRms > 0.2);
97 ASSERT_TRUE(metrics.estimatedSnrDb > 3.0);
98 }
