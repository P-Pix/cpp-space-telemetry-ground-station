96 void testBlockingQueueAndBackpressure()
97 {
98 stgs::BlockingQueue<int> queue(1U);
99 ASSERT_TRUE(queue.push(1));
100 auto blockedProducer = std::async(std::launch::async, [&queue]
101 { return queue.push(2); });
102 ASSERT_EQ(blockedProducer.wait_for(std::chrono::milliseconds(20)), std::future_status::timeout);
103 const auto first = queue.pop();
104 ASSERT_TRUE(first.has_value());
105 ASSERT_EQ(*first, 1);
106 ASSERT_EQ(blockedProducer.wait_for(std::chrono::seconds(1)), std::future_status::ready);
107 ASSERT_TRUE(blockedProducer.get());
108 queue.close();
109 const auto second = queue.pop();
110 ASSERT_TRUE(second.has_value());
111 ASSERT_EQ(*second, 2);
112 ASSERT_TRUE(!queue.pop().has_value());
