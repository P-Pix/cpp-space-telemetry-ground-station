60 void testNonFiniteTemperatureRejected()
61 {
62 auto frame = sampleFrame();
63 frame.temperatureC = std::numeric_limits<float>::infinity();
64 assertThrows([&]
65 { (void)stgs::encodeFrame(frame); });
66
67 // Construire d'abord une trame valide, puis remplacer les bits de température par +Inf et
68 // recalculer le CRC. Le rejet vérifie donc la règle sémantique, pas une corruption du checksum.
69 frame.temperatureC = 20.0F;
70 auto bytes = stgs::encodeFrame(frame);
71 const auto positiveInfinityBits = std::bit_cast<std::uint32_t>(std::numeric_limits<float>::infinity());
72 stgs::ByteVector encodedInfinity;
73 stgs::detail::appendU32BE(encodedInfinity, positiveInfinityBits);
74 std::copy(encodedInfinity.begin(), encodedInfinity.end(),
75 bytes.begin() + static_cast<std::ptrdiff_t>(stgs::TemperatureOffset));
76
77 bytes.resize(bytes.size() - stgs::CrcSize);
78 const auto crc = stgs::crc32(bytes);
79 stgs::detail::appendU32BE(bytes, crc);
80
81 const auto parsed = stgs::decodeFrame(bytes);
82 ASSERT_TRUE(std::holds_alternative<stgs::FrameError>(parsed));
83 ASSERT_EQ(std::get<stgs::FrameError>(parsed).code, stgs::FrameErrorCode::InvalidTemperature);
84 }

102

105 chunk1.insert(chunk1.end(), frameA.begin(), frameA.begin() + 10);
106 auto output = extractor.feed(chunk1);
107 ASSERT_TRUE(output.empty());
108
