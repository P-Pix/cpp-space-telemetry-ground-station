137 stgs::TelemetryFrame makeFrame(const Options &opts, std::mt19937 &rng, std::size_t index)
138 {
139 std::uniform_int_distribution<unsigned int> byteDist(
140 0U, static_cast<unsigned int>(std::numeric_limits<std::uint8_t>::max()));
141
142 stgs::TelemetryFrame frame;
143 const auto satelliteOffset = index % BenchmarkSatellitePoolSize;
144 frame.satelliteId = static_cast<std::uint16_t>(
145 static_cast<std::size_t>(BenchmarkSatelliteBaseId) + satelliteOffset);
146 frame.timestampMs = BenchmarkTimestampBaseMs + static_cast<std::uint64_t>(index);
147 frame.temperatureC = BenchmarkTemperatureBaseC +
148 static_cast<float>(index % BenchmarkTemperatureCycleSteps) * BenchmarkTemperatureStepC;
149 frame.batteryPercent = static_cast<std::uint8_t>(
150 static_cast<std::size_t>(stgs::MaxBatteryPercent) - (index % BenchmarkBatteryCycleStates));
151 frame.status = stgs::Status::Nominal;
152 frame.payload.resize(opts.payloadSize);
153 for (auto &byte : frame.payload)
154 {
155 byte = static_cast<std::uint8_t>(byteDist(rng));
156 }
157 return frame;
158 }

162int main(int argc, char **argv)
163{
164 try
165 {
166 const auto opts = parseArgs(argc, argv);
167 std::mt19937 rng(opts.seed);
168 std::vector<stgs::ByteVector> encoded;
169 encoded.reserve(opts.frames);
170 for (std::size_t i = 0; i < opts.frames; ++i)
171 {
172 encoded.push_back(stgs::encodeFrame(makeFrame(opts, rng, i)));
173 }
174
175 stgs::BlockingQueue<stgs::ByteVector> queue(BenchmarkQueueCapacity);
176 std::atomic_size_t decoded{0};
177 std::atomic_size_t rejected{0};
178
179 std::vector<std::thread> workers;
180 workers.reserve(opts.decoderThreads);
181 const auto start = std::chrono::steady_clock::now();
182 for (std::size_t i = 0; i < opts.decoderThreads; ++i)
183 {
184 workers.emplace_back([&]
185 {
186 while (auto bytes = queue.pop()) {
187 auto result = stgs::decodeFrame(*bytes);
188 if (std::holds_alternative<stgs::TelemetryFrame>(result)) {
189 ++decoded;
190 } else {
191 ++rejected;
192 }
193 } });
194 }
195
196 for (const auto &frame : encoded)
197 {
198 queue.push(frame);
199 }
200 queue.close();
201 for (auto &worker : workers)
202 {
203 worker.join();
204 }
205 const auto end = std::chrono::steady_clock::now();
206 const auto seconds = std::chrono::duration<double>(end - start).count();
207 const auto fps = static_cast<double>(decoded.load() + rejected.load()) / seconds;
208
209 std::cout << "benchmark=decode_pipeline"
210 << " frames=" << opts.frames
211 << " payload_size=" << opts.payloadSize
212 << " decoder_threads=" << opts.decoderThreads
213 << " decoded=" << decoded.load()
214 << " rejected=" << rejected.load()
215 << " duration_seconds=" << seconds
216 << " frames_per_second=" << fps << '\n';
217
218 if (decoded.load() != opts.frames || rejected.load() != 0U)
219 {
220 return 2;
221 }
222 return 0;
223 }
224 catch (const std::exception &ex)

226 std::cerr << "error: " << ex.what() << '\n';
227 std::cerr << "run with --help for usage\n";
